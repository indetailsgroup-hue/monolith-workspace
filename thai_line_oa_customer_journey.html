🤖 Keyword Auto-reply Rules
Auto-reply ทั้งหมดจัดการผ่าน webhook handler (Sale Agent) — ไม่ใช้ built-in auto-reply ของ LINE OA Manager เพื่อให้ตอบสนองได้ dynamic ตาม context ของแต่ละลูกค้า
Rule 01 — สถานะโครงการ Sale Agent handles
Keywords: สถานะ ความคืบหน้า งานไปถึงไหนแล้ว status

Action: ดึง project status จาก Supabase → ส่ง status_update Flex Message พร้อม progress % และ stage ปัจจุบัน

Rule 02 — ติดต่อพนักงาน Sale Agent → human handoff
Keywords: คุยกับคน ต้องการพนักงาน โทรหาผม urgent ด่วน

Action: ส่ง Flex พร้อมเบอร์โทรตรง + notify Sale team ใน LINE Group ว่ามีลูกค้าขอคุยกับคน

Rule 03 — นัดหมาย Sale Agent → LIFF redirect
Keywords: นัดหมาย จะมาดูหน้างาน วันติดตั้ง เลื่อนนัด

Action: ส่ง URI button → เปิด LIFF schedule page liff.line.me/{LIFF_ID_SCHEDULE}

Rule 04 — ราคา / ใบเสนอราคา Sale Agent → inquiry form
Keywords: ราคา ใบเสนอราคา quotation งบประมาณ ค่าใช้จ่าย

Action: ส่ง Flex พร้อม "กรอกฟอร์มขอใบเสนอราคา" → URI → LIFF inquiry form

Rule 05 — FAQ / ข้อมูลทั่วไป Sale Agent → knowledge base lookup
Keywords: วัสดุ ประกัน warranty ระยะเวลา ผลิตจากอะไร

Action: Sale Agent ค้นหาใน knowledge base → ตอบด้วย text หรือ Flex card พร้อม source reference

Rule 06 — Default (ไม่ match keyword ใด) Sale Agent → NLU fallback
Pattern: ข้อความที่ไม่ตรง keyword ใดๆ

Action: Sale Agent ประมวลผลด้วย LLM → ตอบ text message ทั่วไป → ถ้า confidence ต่ำ → แนบ Rich Card "ให้พนักงานช่วยเหลือ"

import os from linebot.v3.messaging import ( ApiClient, Configuration, MessagingApi, PushMessageRequest, FlexMessage, FlexContainer ) from services.flex_builder import build_flex config = Configuration( access_token=os.getenv("LINE_CHANNEL_ACCESS_TOKEN") ) def notify_customer(line_user_id: str, event_type: str, payload: dict): """Push Flex Message ให้ลูกค้า 1 คน ผ่าน LINE OA""" flex_json = build_flex(event_type, payload) message = FlexMessage( alt_text=flex_json["altText"], contents=FlexContainer.from_dict(flex_json["contents"]) ) with ApiClient(config) as client: api = MessagingApi(client) api.push_message(PushMessageRequest( to=line_user_id, messages=[message] )) def broadcast_to_customers(user_ids: list, event_type: str, payload: dict): """Push Flex ให้หลายคนพร้อมกัน (max 500 per call)""" flex_json = build_flex(event_type, payload) message = FlexMessage( alt_text=flex_json["altText"], contents=FlexContainer.from_dict(flex_json["contents"]) ) # Split into chunks of 500 (LINE API limit) for i in range(0, len(user_ids), 500): chunk = user_ids[i:i+500] with ApiClient(config) as client: api = MessagingApi(client) api.multicast_message( MulticastMessageRequest(to=chunk, messages=[message]) )

const el = (id: string) => document.getElementById(id)!;

function renderStatus(project: Project): void {
  el("project-code").textContent = project.project_code;
  el("stage-label").textContent  = project.stage_label;
  el("updated-at").textContent   =
    new Date(project.updated_at).toLocaleString("th-TH", { timeZone: "Asia/Bangkok" });

  // progress bar (0–4 stages)
  const stages = ["pending", "in_design", "production", "installation", "complete"];
  const pct    = ((stages.indexOf(project.status) + 1) / stages.length) * 100;
  (el("progress-bar") as HTMLElement).style.width = `${pct}%`;
}

// ─── subscribe to Supabase real-time ─────────────────────────────────────────
function subscribeProject(projectId: string): RealtimeChannel {
  return supabase
    .channel(`project-status-${projectId}`)
    .on<Project>(
      "postgres_changes",
      {
        event:  "UPDATE",
        schema: "public",
        table:  "projects",
        filter: `id=eq.${projectId}`,
      },
      (payload: RealtimePayload) => {
        renderStatus(payload.new);
        // pulse animation feedback
        el("status-card").classList.add("pulse");
        setTimeout(() => el("status-card").classList.remove("pulse"), 1000);
      }
    )
    .subscribe();
}

// ─── main init ───────────────────────────────────────────────────────────────
async function init(): Promise<void> {
  // 1. init LIFF
  await liff.init({ liffId: LIFF_ID });

  if (!liff.isLoggedIn()) {
    liff.login({ redirectUri: window.location.href });
    return;
  }

  // 2. get LINE user profile
  const profile = await liff.getProfile();
  el("user-name").textContent = profile.displayName;
  el("avatar").setAttribute("src", profile.pictureUrl ?? "");

  // 3. get projectId from URL path /liff/status/:projectId
  const projectId = window.location.pathname.split("/").pop()!;

  // 4. fetch initial project data from Supabase
  const { data, error } = await supabase
    .from("projects")
    .select("id, project_code, line_user_id, status, stage_label, updated_at")
    .eq("id", projectId)
    .eq("line_user_id", profile.userId)   // security: own projects only
    .single();

  if (error || !data) {
    el("error-msg").textContent = "ไม่พบข้อมูลโครงการ หรือไม่มีสิทธิ์เข้าถึง";
    return;
  }

  renderStatus(data);

  // 5. subscribe to live updates
  subscribeProject(projectId);
}

init().catch(console.error);

หน้า /liff/approval/{project_id} — ลูกค้า review แบบ 3D แล้วกด Approve/Request Change ผ่าน Supabase design_approvals table พร้อม real-time sync 

// liff-approval.ts — LIFF_ID_APPROVAL page
import liff from "@line/liff";
import { createClient, RealtimeChannel } from "@supabase/supabase-js";

// ─── env vars ────────────────────────────────────────────────────────────────
const LIFF_ID = import.meta.env.VITE_LIFF_ID_APPROVAL as string;
const SB_URL  = import.meta.env.VITE_SUPABASE_URL      as string;
const SB_KEY  = import.meta.env.VITE_SUPABASE_ANON_KEY as string;

const supabase = createClient(SB_URL, SB_KEY);

// ─── types ────────────────────────────────────────────────────────────────────
type ApprovalStatus = "pending" | "approved" | "change_requested" | "revision_submitted";

interface DesignApproval {
  id:              string;
  project_id:      string;
  version:         number;
  design_url:      string;
  status:          ApprovalStatus;
  reviewer_notes:  string | null;
  approved_at:     string | null;
  line_user_id:    string;
}

// ─── submit approval action ───────────────────────────────────────────────────
async function submitApproval(
  approvalId: string,
  action:     "approved" | "change_requested",
  notes:      string
): Promise<void> {
  const update: Partial<DesignApproval> = {
    status:         action,
    reviewer_notes: notes || null,
    approved_at:    action === "approved" ? new Date().toISOString() : null,
  };

  const { error } = await supabase
    .from("design_approvals")
    .update(update)
    .eq("id", approvalId);

  if (error) throw new Error(error.message);

  // close LIFF window after action
  await new Promise(r => setTimeout(r, 800));
  liff.closeWindow();
}

// ─── real-time: watch for Designer revisions ─────────────────────────────────
function subscribeApproval(approvalId: string): RealtimeChannel {
  return supabase
    .channel(`approval-${approvalId}`)
    .on<DesignApproval>(
      "postgres_changes",
      {
        event:  "UPDATE",
        schema: "public",
        table:  "design_approvals",
        filter: `id=eq.${approvalId}`,
      },
      (payload) => {
        const rec = payload.new;
        if (rec.status === "revision_submitted") {
          // Designer uploaded a new revision — reload preview
          (document.getElementById("design-frame") as HTMLIFrameElement).src =
            rec.design_url + "?v=" + rec.version;
          document.getElementById("version-badge")!.textContent =
            `Rev ${rec.version}`;
          document.getElementById("action-panel")!.style.display = "block";
        }
      }
    )
    .subscribe();
}

// ─── main init ───────────────────────────────────────────────────────────────
async function init(): Promise<void> {
  await liff.init({ liffId: LIFF_ID });

  if (!liff.isLoggedIn()) {
    liff.login({ redirectUri: window.location.href });
    return;
  }

  const profile   = await liff.getProfile();
  const projectId = window.location.pathname.split("/").pop()!;

  // fetch latest pending approval for this project + user
  const { data: approval, error } = await supabase
    .from("design_approvals")
    .select("*")
    .eq("project_id", projectId)
    .eq("line_user_id", profile.userId)
    .in("status", ["pending", "revision_submitted"])
    .order("version", { ascending: false })
    .limit(1)
    .single();

  if (error || !approval) {
    document.getElementById("error-msg")!.textContent =
      "ไม่มีแบบที่รอการอนุมัติในขณะนี้";
    return;
  }

  // render design preview + version badge
  (document.getElementById("design-frame") as HTMLIFrameElement).src =
    approval.design_url;
  document.getElementById("version-badge")!.textContent =
    `Rev ${approval.version}`;
  document.getElementById("user-name")!.textContent = profile.displayName;

  // bind approve / request-change buttons
  document.getElementById("btn-approve")!.addEventListener("click", async () => {
    await submitApproval(approval.id, "approved", "");
  });

  document.getElementById("btn-change")!.addEventListener("click", async () => {
    const notes = (document.getElementById("change-notes") as HTMLTextAreaElement).value;
    if (!notes.trim()) {
      alert("กรุณาระบุรายละเอียดที่ต้องการแก้ไข");
      return;
    }
    await submitApproval(approval.id, "change_requested", notes);
  });

  // subscribe to Designer revisions in real-time
  subscribeApproval(approval.id);
}

init().catch(console.error);
