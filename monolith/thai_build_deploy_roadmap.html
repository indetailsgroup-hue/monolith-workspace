🎯 4 การตัดสินใจสำคัญก่อนเริ่ม Build
1. ใช้ LLM Provider ใด?
แนะนำGPT-4o — Vision ดีที่สุด, Thai language ดี, API เสถียร, ราคาสมเหตุสมผล
ทางเลือกClaude 3.5 — reasoning ดีมาก, context ยาว, Vision ดี แต่ราคาสูงกว่า
2. Multi-agent vs Single Router?
แนะนำMaster Router + 28 Specialists — user ส่ง input ครั้งเดียว, router ส่งต่อ agent ที่ถูกต้องตาม SOP
ทางเลือกSingle LLM + function calling — ง่ายกว่าแต่ไม่ scale หาก agents เพิ่ม
3. Build in-house vs ใช้ Partner?
แนะนำHybrid — hire AI Engineer 1-2 คน + ใช้ SOP เป็น spec ให้ vendor พัฒนา Phase C-D
ทางเลือกFull outsource — เร็วกว่าแต่ต้องควบคุม prompt quality ด้วยตัวเอง
4. เริ่ม Deploy กี่ Agent ก่อน?
แนะนำ5 agents แรก — เลือก high-impact + เรียนรู้ pattern ก่อน scale ไป 28
ทางเลือก14 agents (Phase 1+2) — deploy ตาม phase ที่ SOP กำหนด

-- Enable pgvector extension
CREATE EXTENSION IF NOT EXISTS vector;

CREATE TABLE IF NOT EXISTS knowledge_embeddings (
  id               uuid          PRIMARY KEY DEFAULT gen_random_uuid(),
  source_id        text          NOT NULL,                              -- 'daph-second-brain'
  source_version   text          NOT NULL DEFAULT 'daph-qms-2020-rev00',
  imported_at      timestamptz   NOT NULL,
  review_status    text          NOT NULL DEFAULT 'draft',             -- 'draft' | 'approved'
  process_step     text,                                               -- step nameEN (e.g. 'Sale', 'Designer')
  sub_process_group text,                                              -- 'Office' | 'Factory' | 'Installation'
  content          text          NOT NULL,                             -- chunk text สำหรับ embed
  content_type     text          NOT NULL                              -- ประเภทเนื้อหา
                   CHECK (content_type IN (
                     'process_step',   -- 28 steps จาก processModel
                     'pfmea_row',      -- 122 rows จาก pfmeaRiskRows
                     'raci_entry'      -- RACI สำหรับแต่ละ step
                   )),
  embedding        vector(1536),                                       -- text-embedding-3-small dims
  metadata         jsonb         DEFAULT '{}',                        -- canonOrder, severity, detection ฯลฯ
  created_at       timestamptz   NOT NULL DEFAULT now()
);

-- IVFFlat index สำหรับ cosine similarity search
CREATE INDEX IF NOT EXISTS idx_knowledge_embeddings_ivfflat
  ON knowledge_embeddings
  USING ivfflat (embedding vector_cosine_ops)
  WITH (lists = 100);

CREATE INDEX IF NOT EXISTS idx_knowledge_content_type  ON knowledge_embeddings (content_type);
CREATE INDEX IF NOT EXISTS idx_knowledge_review_status ON knowledge_embeddings (review_status);

⚙️ 5 ขั้นตอน Embed daph-second-brain → Supabase
1
Parse _knowledge-export.json
อ่าน 3 arrays: processModel (28 steps), pfmeaRiskRows (122 rows), raciMap (28 entries) · ตรวจสอบ schemaVersion = "1.0.0" และ reviewStatus = "draft"
2
Chunk เนื้อหาเป็น Embedded Documents
แปลงแต่ละ record เป็น chunk text — process_step: "Process Step: {nameEN} ({nameTH}) | Group: {group} | Order: {canonOrder}" · pfmea_row: "PFMEA [{processStep}] Mode: {failureMode} | Sev: {severity} | Controls: {controls}" · รวม 178 chunks
3
OpenAI text-embedding-3-small (1536 dims)
batch 100 chunks/request · estimated cost: 178 chunks × avg 40 tokens = ~7,120 tokens ≈ $0.0001 · model: text-embedding-3-small · output: float32[1536]
4
Upsert เข้า Supabase knowledge_embeddings
upsert on conflict (source_id, source_version, content_type, process_step) · บันทึก review_status = "draft" ทุก row เพื่อให้ MCP Layer ส่ง low_confidence_warning: true กลับ agent
5
Verify Count & Spot-check

💻 TypeScript Embed Script — scripts/embed-knowledge.ts
อ้างอิงจาก field จริงใน _knowledge-export.json (schemaVersion 1.0.0)
import { createClient } from '@supabase/supabase-js';
import OpenAI from 'openai';
import knowledgeExport from './_knowledge-export.json';

// Types จาก schemaVersion 1.0.0
interface ProcessStep {
  canonOrder: number; nameEN: string; nameTH: string;
  group: 'Office' | 'Factory' | 'Installation';
  requiresApproval: boolean;
}
interface PfmeaRow {
  id: string; processStep: string; failureMode: string;
  effect: string; severity: number; cause: string;
  detection: number; controls: string[];
}
interface RaciEntry {
  processStep: string; responsible: string[];
  accountable: string; consulted: string[];
  informed: string[]; confidence: number;
}

const supabase = createClient(
  process.env.SUPABASE_URL!, process.env.SUPABASE_ANON_KEY!
);
const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY! });

type EmbedDoc = {
  content: string; content_type: string;
  process_step: string; sub_process_group: string; metadata: object;
};

async function buildChunks(): Promise<EmbedDoc[]> {
  const docs: EmbedDoc[] = [];

  // 1. processModel → process_step chunks (28 steps)
  for (const step of knowledgeExport.processModel as ProcessStep[]) {
    docs.push({
      content: `Process Step: ${step.nameEN} (${step.nameTH}) | Group: ${step.group} | Order: ${step.canonOrder} | RequiresApproval: ${step.requiresApproval}`,
      content_type: 'process_step',
      process_step: step.nameEN,
      sub_process_group: step.group,
      metadata: { canonOrder: step.canonOrder, requiresApproval: step.requiresApproval }
    });
  }

  // 2. pfmeaRiskRows → pfmea_row chunks (122 rows)
  for (const row of knowledgeExport.pfmeaRiskRows as PfmeaRow[]) {
    docs.push({
      content: `PFMEA [${row.processStep}] FailureMode: ${row.failureMode} | Effect: ${row.effect} | Sev: ${row.severity} | Controls: ${row.controls.join('; ')}`,
      content_type: 'pfmea_row',
      process_step: row.processStep,
      sub_process_group: '',
      metadata: { id: row.id, severity: row.severity, detection: row.detection, cause: row.cause }
    });
  }

  // 3. raciMap → raci_entry chunks (28 entries)
  for (const raci of knowledgeExport.raciMap as RaciEntry[]) {
    docs.push({
      content: `RACI [${raci.processStep}] Responsible: ${raci.responsible.join(', ')} | Accountable: ${raci.accountable} | Consulted: ${raci.consulted.join(', ')} | Informed: ${raci.informed.join(', ')}`,
      content_type: 'raci_entry',
      process_step: raci.processStep,
      sub_process_group: '',
      metadata: { confidence: raci.confidence }
    });
  }
  return docs;
}

async function embedAndUpsert() {
  const docs = await buildChunks();
  console.log(`Built ${docs.length} chunks (expected: 178)`);

  // batch embed — 100 per API call
  const BATCH = 100;
  for (let i = 0; i < docs.length; i += BATCH) {
    const batch = docs.slice(i, i + BATCH);
    const resp = await openai.embeddings.create({
      model: 'text-embedding-3-small', input: batch.map(d => d.content)
    });
    const rows = batch.map((doc, j) => ({
      source_id: 'daph-second-brain',
      source_version: knowledgeExport.sourceVersion,
      imported_at: knowledgeExport.importedAt,
      review_status: knowledgeExport.reviewStatus,   // 'draft'
      process_step: doc.process_step,
      sub_process_group: doc.sub_process_group,
      content: doc.content,
      content_type: doc.content_type,
      embedding: resp.data[j].embedding,
      metadata: doc.metadata
    }));
    const { error } = await supabase.from('knowledge_embeddings').upsert(rows);
    if (error) throw error;
    console.log(`Upserted batch ${i / BATCH + 1}`);
  }
  console.log('✅ Done — verify with:');
  console.log("SELECT content_type, COUNT(*) FROM knowledge_embeddings GROUP BY content_type;");
}

embedAndUpsert().catch(console.error);
🔍 RAG Query Pattern — ใช้ใน MCP Tool query_knowledge
-- rpc_mcp_query_knowledge (Supabase RPC function)
CREATE OR REPLACE FUNCTION rpc_mcp_query_knowledge(
  query_embedding  vector(1536),
  process_step_filter text  DEFAULT NULL,
  content_type_filter text  DEFAULT NULL,
  match_count int           DEFAULT 5
) RETURNS TABLE (
  content text, content_type text, process_step text,
  review_status text, similarity float, metadata jsonb
) LANGUAGE sql AS $$
  SELECT content, content_type, process_step, review_status,
         1 - (embedding <<=> query_embedding) AS similarity, metadata
  FROM knowledge_embeddings
  WHERE (process_step_filter IS NULL OR process_step = process_step_filter)
    AND   (content_type_filter IS NULL OR content_type = content_type_filter)
  ORDER BY embedding <=> query_embedding
  LIMIT match_count;
$$;
